4. Zoneamento Cadastral Trifásico de Vilas

O modelo de 3 anéis concêntricos que protege assentamentos contra destruição por estradas e obras.

O Modelo dos 3 Anéis Concêntricos

Todas as vilas e estruturas catalogadas pelo Rota Core são representadas através da classe imutável SettlementZone, dividida em 3 camadas de segurança:

  1. Reserva Estrutural (Borda Roxa - structuralBounds): A caixa delimitadora original gerada pelo Minecraft/CTOV.
  2. Núcleo Urbano (Borda Verde - urbanCoreBounds): O polígono justo que envolve estritamente as casas habitadas, camas, praça central e sinos. Área sagrada: nenhum bloco aqui pode ser quebrado por estradas.
  3. Perímetro de Aproximação (Borda Amarela - approachPerimeterBounds): Faixa externa de segurança (8 a 12 blocos de recuo). É aqui onde as estradas devem parar e ancorar nos portais viários homologados, sem rasgar a vila ao meio.

1. Consultando uma Vila no Mundo

package com.exemplo.mod;

import com.rotacore.world.zoning.SettlementZone;
import com.rotacore.world.zoning.StructureZoneRegistry;
import net.minecraft.core.BlockPos;
import net.minecraft.server.level.ServerLevel;

public class ServicoDeObras {
    public void verificarProximidade(ServerLevel level, BlockPos pos) {
        // Consulta no Core em O(1) sem lag
        SettlementZone vilaProxima = StructureZoneRegistry.getZoneAt(level, pos);

        if (vilaProxima != null) {
            if (vilaProxima.isInsideUrbanCore(pos)) {
                // Bloqueia qualquer obra física aqui
                System.out.println("Posição dentro das casas da vila: " + vilaProxima.getName());
            } else if (vilaProxima.isInsideApproachPerimeter(pos)) {
                System.out.println("Posição no perímetro externo de aproximação viária!");
            }
        }
    }
}
